function doPost(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName("Orders"); // Auto-create the sheet if you forgot to rename it if (!sheet) { sheet = ss.insertSheet("Orders"); } // Add Headers if empty if (sheet.getLastRow() === 0) { sheet.appendRow(["OrderNumber", "Timestamp", "Customer", "DeliveryDate", "CartData"]); } const data = JSON.parse(e.postData.contents); const { orderNumber, timestamp, customer, deliveryDate, cart } = data; const cartStr = JSON.stringify(cart); const rows = sheet.getDataRange().getValues(); let rowIndex = -1; // Check if modifying an existing order for(let i = 0; i < rows.length; i++) { if(rows[i][0] === orderNumber) { rowIndex = i + 1; break; } } if(rowIndex > -1) { sheet.getRange(rowIndex, 1, 1, 5).setValues([[orderNumber, timestamp, customer, deliveryDate, cartStr]]); } else { sheet.appendRow([orderNumber, timestamp, customer, deliveryDate, cartStr]); } return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({error: err.toString()})).setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Orders"); if (!sheet) return ContentService.createTextOutput("[]").setMimeType(ContentService.MimeType.JSON); const rows = sheet.getDataRange().getValues(); const history = []; for(let i = 1; i < rows.length; i++) { if(rows[i][0]) { try { history.push({ orderNumber: String(rows[i][0]), timestamp: String(rows[i][1]), customer: String(rows[i][2]), deliveryDate: String(rows[i][3]), cart: JSON.parse(rows[i][4]) }); } catch(e) {} // skip broken row data safely } } return ContentService.createTextOutput(JSON.stringify(history)).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput("[]").setMimeType(ContentService.MimeType.JSON); } }
